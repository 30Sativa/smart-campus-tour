// Lenis smooth scroll - chỉ dùng gsap.ticker, không dùng RAF riêng
const lenis = new Lenis({
  duration: 1.2,
  easing: t => Math.min(1, 1.001 - Math.pow(2, -10 * t)),
  smoothWheel: true,
  wheelMultiplier: 1,
  touchMultiplier: 2
});
gsap.ticker.add(time => lenis.raf(time * 1000));
gsap.ticker.lagSmoothing(0);
lenis.on("scroll", ScrollTrigger.update);

gsap.registerPlugin(ScrollTrigger);
(function(){
  const img=document.querySelector("#hero-img img");
  const t1=document.getElementById("ht1"),t2=document.getElementById("ht2");
  const tags=document.getElementById("htags"),scrl=document.getElementById("hscroll"),stat=document.getElementById("hstat");
  function split(el){if(!el)return[];const txt=el.textContent.trim();el.textContent="";const ls=[];[...txt].forEach(c=>{const s=document.createElement("span");s.textContent=c;s.style.cssText="display:inline-block;will-change:transform,opacity,filter";el.appendChild(s);ls.push(s);});return ls;}
  const l1=split(t1),l2=split(t2);
  gsap.set(img,{scale:1.15,filter:"blur(20px)",opacity:0});
  gsap.set([...l1,...l2],{opacity:0,filter:"blur(14px)",y:36});
  gsap.set([tags,scrl,stat],{opacity:0,y:18});
  const tl=gsap.timeline({defaults:{ease:"power3.out"}});
  tl.to(img,{scale:1.05,filter:"blur(0px)",opacity:1,duration:2.2,ease:"power2.out"})
    .to(l1,{opacity:1,filter:"blur(0px)",y:0,duration:1.1,stagger:.06},"-=1.8")
    .to(l2,{opacity:1,filter:"blur(0px)",y:0,duration:1.1,stagger:.06},"-=0.7")
    .to([tags,stat],{opacity:1,y:0,duration:.8,stagger:.1},"-=0.4")
    .to(scrl,{opacity:1,y:0,duration:.6},"-=0.2");
  tl.call(()=>gsap.to(img,{scale:1.12,duration:10,ease:"sine.inOut",yoyo:true,repeat:-1}));
  ScrollTrigger.create({trigger:"#hero",start:"top top",end:"bottom top",onUpdate:s=>gsap.set(img,{y:s.progress*160})});
})();

(function(){
  const slides=[...document.querySelectorAll(".how-slide")];
  const track=document.getElementById("how-track");
  let cur=0,busy=false,auto;
  function center(i,anim=true){const vw=window.innerWidth,sl=slides[i],off=vw/2-(sl.offsetLeft+sl.offsetWidth/2);gsap.to(track,{x:off,duration:anim?.8:0,ease:"power3.inOut",overwrite:true});slides.forEach((s,j)=>s.classList.toggle("on",j===i));}
  function go(d){if(busy)return;busy=true;cur=Math.max(0,Math.min(slides.length-1,cur+d));center(cur);setTimeout(()=>{busy=false;},850);}
  document.getElementById("how-prev").addEventListener("click",()=>go(-1));
  document.getElementById("how-next").addEventListener("click",()=>go(1));
  requestAnimationFrame(()=>center(cur,false));
  window.addEventListener("resize",()=>center(cur,false));
  auto=setInterval(()=>go(cur<slides.length-1?1:-(slides.length-1)),3800);
  const wrap=document.getElementById("how-slides");
  wrap.addEventListener("mouseenter",()=>clearInterval(auto));
  wrap.addEventListener("mouseleave",()=>{auto=setInterval(()=>go(cur<slides.length-1?1:-(slides.length-1)),3800);});
})();

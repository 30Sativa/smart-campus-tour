<script>
  window.Webflow ||= [];
  window.Webflow.push(() => {

    const train = document.querySelector(".rt-design-v1-slider-train");
    if (!train) {
      console.warn("Slider: .rt-design-v1-slider-train not found");
      return;
    }
    if (typeof gsap === "undefined") {
      console.warn("Slider: GSAP not loaded — check script order (GSAP CDN must load BEFORE this script)");
      return;
    }

    const original = [...train.querySelectorAll(".rt-design-v1-slider-block")];
    if (!original.length) {
      console.warn("Slider: no .rt-design-v1-slider-block elements found");
      return;
    }

    gsap.set(train, { opacity: 0 });

    const viewport = train.parentElement;
    const total = original.length;

    original.slice().reverse().forEach(el => train.prepend(el.cloneNode(true)));
    original.forEach(el => train.appendChild(el.cloneNode(true)));

    const slides = [...train.querySelectorAll(".rt-design-v1-slider-block")];
    let current = total;
    let busy = false;
    let autoplay = null;

    function moveX(index) {
      const slide = slides[index];
      return viewport.clientWidth / 2 - (slide.offsetLeft + slide.offsetWidth / 2);
    }

    function states(anim = true) {
      slides.forEach((slide, i) => {
        const active = i === current;
        const content = slide.querySelector(".rt-design-v1-slider-content");
        gsap.to(slide, {
          scale: active ? 1 : .72,
          opacity: active ? 1 : .4,
          duration: anim ? .6 : 0,
          ease: "power3.out",
          overwrite: true
        });
        if (content) gsap.to(content, {
          opacity: active ? 1 : 0,
          duration: anim ? .45 : 0,
          ease: "power2.out",
          overwrite: true
        });
      });
    }

    function go(dir) {
      if (busy) return;
      busy = true;
      current += dir;

      gsap.to(train, {
        x: moveX(current),
        duration: .8,
        ease: "power3.inOut",
        overwrite: true,
        onComplete: () => {
          if (current >= total * 2) {
            current = total;
            gsap.set(train, { x: moveX(current) });
          }
          if (current < total) {
            current = total * 2 - 1;
            gsap.set(train, { x: moveX(current) });
          }
          states(false);
          busy = false;
        }
      });

      states(true);
    }

    function startAutoplay() {
      stopAutoplay();
      autoplay = setInterval(() => go(1), 3000);
    }

    function stopAutoplay() {
      if (autoplay) clearInterval(autoplay);
    }

    /* INITIAL */
    gsap.set(slides, { scale: .72, opacity: .4 });
    gsap.set(train.querySelectorAll(".rt-design-v1-slider-content"), { opacity: 0 });

    requestAnimationFrame(() => {
      gsap.set(train, { x: moveX(current) });
      states(false);
      gsap.to(train, { opacity: 1, duration: .3, ease: "power1.out" });
      startAutoplay();
    });

    /* RESIZE */
    window.addEventListener("resize", () => {
      gsap.set(train, { x: moveX(current) });
    });

    /* OPTIONAL: pause on hover */
    viewport.addEventListener("mouseenter", stopAutoplay);
    viewport.addEventListener("mouseleave", startAutoplay);

  });
</script>

<script>
{
  const items = [...document.querySelectorAll('[service-one="item"]')];
  const images = [...document.querySelectorAll('[service-image="item"]')];
  let active = 0, timer;

  function getTextEls(item) {
    const els = [item.querySelector(".services-title"), item.querySelector(".services-title-clone")];
    return els.filter(Boolean);
  }

  items.forEach((item, i) => {
    gsap.set(getTextEls(item), { opacity: i === 0 ? 1 : 0.7 });
    if (images[i]) gsap.set(images[i], {
      opacity: i === 0 ? 1 : 0,
      scale: i === 0 ? 1 : 1.08,
      clipPath: i === 0 ? "inset(0%)" : "inset(100% 0% 0% 0%)",
      zIndex: i === 0 ? 2 : 1
    });
    item.addEventListener("mouseenter", () => {
      clearTimeout(timer);
      change(i);
      timer = setTimeout(loop, 3500);
    });
  });

  function change(index) {
    if (index === active) return;

    gsap.to(getTextEls(items[active]), { opacity: 0.7, duration: .5, ease: "power2.out", overwrite: true });
    gsap.to(getTextEls(items[index]), { opacity: 1, duration: .5, ease: "power2.out", overwrite: true });

    const oldImg = images[active], nextImg = images[index];
    if (oldImg && nextImg) {
      gsap.set(nextImg, { opacity: 0, scale: 1.08, clipPath: "inset(100% 0% 0% 0%)", zIndex: 3 });
      gsap.to(oldImg, { opacity: 0, scale: 1.02, clipPath: "inset(0% 0% 100% 0%)", duration: .7, ease: "power2.inOut", overwrite: true });
      gsap.to(nextImg, {
        opacity: 1, scale: 1, clipPath: "inset(0%)", duration: .9, ease: "expo.out", overwrite: true,
        onComplete: () => {
          gsap.set(nextImg, { opacity: 1, scale: 1, clipPath: "inset(0%)", zIndex: 2 });
          gsap.set(oldImg, { opacity: 0, scale: 1, clipPath: "inset(100% 0% 0% 0%)", zIndex: 1 });
        }
      });
    }

    active = index;
  }

  function loop() {
    change((active + 1) % items.length);
    timer = setTimeout(loop, 3500);
  }
  timer = setTimeout(loop, 3500);
}
</script>

      function splitLettersByWord(el) {
        const text = el.textContent;

        el.textContent = "";

        const allLetters = [];

        words.forEach((word, wordIndex) => {
          const wordSpan = document.createElement("span");

          [...word].forEach(char => {

          });

          el.appendChild(wordSpan);

          if (wordIndex < words.length - 1) {
            el.appendChild(document.createTextNode(" "));
          }
        });

        return allLetters;
      }

    // Bottom line → center + rotate

      y: -8.5,
      rotate: -45
    }, 0);


  let currentTween;


  // -----------------------------
  // Smooth Sync
  // -----------------------------

  function syncNavAnimation() {

    const isOpen = navButton.classList.contains("w--open");


    // Kill previous animation to prevent overlap
    if (currentTween) {
      currentTween.kill();
    }


    // Desktop → instantly reset
    if (window.innerWidth > 991) {

      currentTween = navMenuTl.tweenTo(0, {
        duration: 0.45,
        ease: "power2.inOut"
      });

      return;
    }


    // Mobile / Tablet
    currentTween = navMenuTl.tweenTo(
      isOpen ? navMenuTl.duration() : 0,
      {
        duration: 0.55,
        ease: "power3.inOut"
      }
    );

  }


  // -----------------------------
  // Observe Webflow Nav State
  // -----------------------------

  const observer = new MutationObserver(syncNavAnimation);

  observer.observe(navButton, {
    attributes: true,
    attributeFilter: ["class"]
  });


  // -----------------------------
  // Resize
  // -----------------------------

  window.addEventListener("resize", () => {

    if (window.innerWidth > 991) {

<script>
  window.Webflow ||= [];
  window.Webflow.push(() => {

    if (typeof gsap === "undefined") {
      console.warn("Hero animation: GSAP not loaded");
      return;
    }

    const image = document.querySelector('[hero-one="image-one"]');
    const titleOne = document.querySelector('[hero-one-bottom="title-one"]');
    const titleTwo = document.querySelector('[hero-one-bottom="title-two"]');
    const textWrap = document.querySelectorAll('[hero-one-bottom="text-wrap"]');
    const scrollBtn = document.querySelector('[hero-one-scroll="trigger"]');

    if (!image) {
      console.warn("Hero animation: image-one not found");
      return;
    }

    function splitLetters(el) {
      if (!el) return [];
      const text = el.textContent.trim();
      const words = text.split(/\s+/);
      el.textContent = "";
      const letters = [];

      words.forEach((word, wordIndex) => {
        const wordWrap = document.createElement("span");
        wordWrap.style.display = "inline-block";
        wordWrap.style.whiteSpace = "nowrap";

        [...word].forEach(char => {
          const letter = document.createElement("span");
          letter.textContent = char;
          letter.style.display = "inline-block";
          letter.style.willChange = "transform, opacity, filter";
          wordWrap.appendChild(letter);
          letters.push(letter);
        });

        el.appendChild(wordWrap);

        if (wordIndex < words.length - 1) {
          el.appendChild(document.createTextNode(" "));
        }
      });

      return letters;
    }

    const titleOneLetters = splitLetters(titleOne);
    const titleTwoLetters = splitLetters(titleTwo);

    /* ---------- responsive tuning ---------- */
    const isMobile = window.matchMedia("(max-width: 479px)").matches;
    const isTablet = window.matchMedia("(max-width: 991px)").matches;

    const imageBlur = isMobile ? "14px" : isTablet ? "18px" : "25px";
    const letterBlur = isMobile ? "10px" : "18px";
    const imageDuration = isMobile ? 1.8 : 2.6;
    const letterStagger = isMobile ? 0.05 : 0.08;

    gsap.set(image, {
      scale: 1.35,
      filter: `blur(${imageBlur})`,
      opacity: 0
    });
    gsap.set(titleOneLetters, {
      opacity: 0,
      filter: `blur(${letterBlur})`,
      y: 30
    });
    gsap.set(titleTwoLetters, {
      opacity: 0,
      filter: `blur(${letterBlur})`,
      y: 30
    });
    if (textWrap.length) {
      gsap.set(textWrap, { opacity: 0, y: 20 });
    }
    if (scrollBtn) {
      gsap.set(scrollBtn, { opacity: 0, y: 30 });
    }

    const tl = gsap.timeline({ defaults: { ease: "power3.out" } });

    tl.to(image, {
      scale: 1.05,
      filter: "blur(0px)",
      opacity: 1,
      duration: imageDuration,
      ease: "power2.out"
    })
    .to(titleOneLetters, {
      opacity: 1,
      filter: "blur(0px)",
      y: 0,
      duration: 1.4,
      stagger: { each: letterStagger, ease: "power1.in" },
      ease: "power3.out"
    }, "-=1.3")
    .to(titleTwoLetters, {
      opacity: 1,
      filter: "blur(0px)",
      y: 0,
      duration: 1.4,
      stagger: { each: letterStagger, ease: "power1.in" },
      ease: "power3.out"
    }, "-=0.9")
    .to(textWrap, {
      opacity: 1,
      y: 0,
      duration: 1,
      ease: "power2.out"
    }, "-=0.4")
    .to(scrollBtn, {
      opacity: 1,
      y: 0,
      duration: .8,
      ease: "power2.out"
    }, "-=0.3");

    tl.call(() => {
      gsap.to(image, {

        repeat: -1
      });
    });

      const totalDistance = gsap.utils.random(220, 340);
      const loopDuration = gsap.utils.random(3, 5);
      const startOffset = gsap.utils.random(0, totalDistance);
      const state = {
        y: startOffset
      };
      gsap.to(state, {
        y: "+=" + totalDistance,
        duration: loopDuration,
        ease: "none",
        repeat: -1,
        modifiers: {
          y: gsap.utils.unitize(y => parseFloat(y) % totalDistance)
        },
        onUpdate: () => {
          const progress = state.y / totalDistance;
          const posY = progress * totalDistance - totalDistance / 2;
          const fade = Math.sin(progress * Math.PI);
          gsap.set(img, {
            y: -posY,
            opacity: fade,
            scale: 0.75 + fade * 0.25
          });
        }
      });
    });
  }

<script>
  window.Webflow ||= [];
  window.Webflow.push(() => {

    const train = document.querySelector(".rt-design-v1-slider-train");
    if (!train) {
      console.warn("Slider: .rt-design-v1-slider-train not found");
      return;
    }
    if (typeof gsap === "undefined") {
      console.warn("Slider: GSAP not loaded — check script order (GSAP CDN must load BEFORE this script)");
      return;
    }

    const original = [...train.querySelectorAll(".rt-design-v1-slider-block")];
    if (!original.length) {
      console.warn("Slider: no .rt-design-v1-slider-block elements found");
      return;
    }

    gsap.set(train, { opacity: 0 });

    const viewport = train.parentElement;
    const total = original.length;

    original.slice().reverse().forEach(el => train.prepend(el.cloneNode(true)));
    original.forEach(el => train.appendChild(el.cloneNode(true)));

    const slides = [...train.querySelectorAll(".rt-design-v1-slider-block")];
    let current = total;
    let busy = false;
    let autoplay = null;

    function moveX(index) {
      const slide = slides[index];
      return viewport.clientWidth / 2 - (slide.offsetLeft + slide.offsetWidth / 2);
    }

    function states(anim = true) {
      slides.forEach((slide, i) => {
        const active = i === current;
        const content = slide.querySelector(".rt-design-v1-slider-content");
        gsap.to(slide, {
          scale: active ? 1 : .72,
          opacity: active ? 1 : .4,
          duration: anim ? .6 : 0,
          ease: "power3.out",
          overwrite: true
        });
        if (content) gsap.to(content, {
          opacity: active ? 1 : 0,
          duration: anim ? .45 : 0,
          ease: "power2.out",
          overwrite: true
        });
      });
    }

    function go(dir) {
      if (busy) return;
      busy = true;
      current += dir;

      gsap.to(train, {
        x: moveX(current),
        duration: .8,
        ease: "power3.inOut",
        overwrite: true,
        onComplete: () => {
          if (current >= total * 2) {
            current = total;
            gsap.set(train, { x: moveX(current) });
          }
          if (current < total) {
            current = total * 2 - 1;
            gsap.set(train, { x: moveX(current) });
          }
          states(false);
          busy = false;
        }
      });

      states(true);
    }

    function startAutoplay() {
      stopAutoplay();
      autoplay = setInterval(() => go(1), 3000);
    }

    function stopAutoplay() {
      if (autoplay) clearInterval(autoplay);
    }

    /* INITIAL */
    gsap.set(slides, { scale: .72, opacity: .4 });
    gsap.set(train.querySelectorAll(".rt-design-v1-slider-content"), { opacity: 0 });

    requestAnimationFrame(() => {
      gsap.set(train, { x: moveX(current) });
      states(false);
      gsap.to(train, { opacity: 1, duration: .3, ease: "power1.out" });
      startAutoplay();
    });

    /* RESIZE */
    window.addEventListener("resize", () => {
      gsap.set(train, { x: moveX(current) });
    });

    /* OPTIONAL: pause on hover */
    viewport.addEventListener("mouseenter", stopAutoplay);
    viewport.addEventListener("mouseleave", startAutoplay);

  });
</script>

<script>
{
  const items = [...document.querySelectorAll('[service-one="item"]')];
  const images = [...document.querySelectorAll('[service-image="item"]')];
  let active = 0, timer;

  function getTextEls(item) {
    const els = [item.querySelector(".services-title"), item.querySelector(".services-title-clone")];
    return els.filter(Boolean);
  }

  items.forEach((item, i) => {
    gsap.set(getTextEls(item), { opacity: i === 0 ? 1 : 0.7 });
    if (images[i]) gsap.set(images[i], {
      opacity: i === 0 ? 1 : 0,
      scale: i === 0 ? 1 : 1.08,
      clipPath: i === 0 ? "inset(0%)" : "inset(100% 0% 0% 0%)",
      zIndex: i === 0 ? 2 : 1
    });
    item.addEventListener("mouseenter", () => {
      clearTimeout(timer);
      change(i);
      timer = setTimeout(loop, 3500);
    });
  });

  function change(index) {
    if (index === active) return;

    gsap.to(getTextEls(items[active]), { opacity: 0.7, duration: .5, ease: "power2.out", overwrite: true });
    gsap.to(getTextEls(items[index]), { opacity: 1, duration: .5, ease: "power2.out", overwrite: true });

    const oldImg = images[active], nextImg = images[index];
    if (oldImg && nextImg) {
      gsap.set(nextImg, { opacity: 0, scale: 1.08, clipPath: "inset(100% 0% 0% 0%)", zIndex: 3 });
      gsap.to(oldImg, { opacity: 0, scale: 1.02, clipPath: "inset(0% 0% 100% 0%)", duration: .7, ease: "power2.inOut", overwrite: true });
      gsap.to(nextImg, {
        opacity: 1, scale: 1, clipPath: "inset(0%)", duration: .9, ease: "expo.out", overwrite: true,
        onComplete: () => {
          gsap.set(nextImg, { opacity: 1, scale: 1, clipPath: "inset(0%)", zIndex: 2 });
          gsap.set(oldImg, { opacity: 0, scale: 1, clipPath: "inset(100% 0% 0% 0%)", zIndex: 1 });
        }
      });
    }

    active = index;
  }

  function loop() {
    change((active + 1) % items.length);
    timer = setTimeout(loop, 3500);
  }
  timer = setTimeout(loop, 3500);
}
</script>
